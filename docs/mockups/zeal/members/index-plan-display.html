整形ロジック ─ app/Models/ZealPlan.php に accessor を追加:

// 案A: 枠数を完全に削除
public function getDisplayNameAttribute(): string
{
    $value = $this->name ?? '';
    // 全角・半角の括弧書きで囲まれた N 枠表記を除去
    $value = preg_replace('/[（(]\s*[0-9０-９]+\s*枠\s*[）)]/u', '', $value);
    // 「1枠 通い放題」のような括弧なし N 枠表記を除去
    $value = preg_replace('/[0-9０-９]+\s*枠[\s　]*/u', '', $value);
    return trim($value);
}

// 案B: 枠数バッジを分離して取得
public function getFrameLabelAttribute(): ?string
{
    // 「1枠 通い放題」「（2枠）通い放題」「フリープラン（1枠）」等から先頭/末尾の N 枠を抽出
    if (preg_match('/[0-9０-９]+/u', mb_convert_kana($this->frameRaw(), 'n'), $m)) {
        return $m[0] . '枠';
    }
    return null;
}

private function frameRaw(): string
{
    if (preg_match('/[（(]\s*[0-9０-９]+\s*枠\s*[）)]/u', $this->name ?? '', $m)) return $m[0];
    if (preg_match('/[0-9０-９]+\s*枠/u', $this->name ?? '', $m)) return $m[0];
    return '';
}

public function getNameWithoutFrameAttribute(): string
{
    return $this->display_name;  // 案A の accessor を再利用
}
